-- Analytics tables for CalorieClick admin dashboard
-- Tracks: restaurant views, healthy nearby sessions, user activity

-- 1. Restaurant view tracking (persistent across deploys)
create table if not exists public.restaurant_views (
  id bigint generated by default as identity primary key,
  place_id text,
  chain_key text,
  place_name text,
  area_key text,
  user_id text,
  market text,
  viewed_at timestamptz not null default now()
);

alter table public.restaurant_views enable row level security;
create policy "service_role_full_access_restaurant_views"
on public.restaurant_views for all to service_role using (true) with check (true);

create index if not exists idx_restaurant_views_chain on public.restaurant_views(chain_key);
create index if not exists idx_restaurant_views_area on public.restaurant_views(area_key);
create index if not exists idx_restaurant_views_date on public.restaurant_views(viewed_at);

-- 2. Healthy Nearby session tracking
create table if not exists public.healthy_nearby_sessions (
  id bigint generated by default as identity primary key,
  user_id text,
  lat numeric,
  lng numeric,
  area_key text,
  places_returned integer default 0,
  top_place_name text,
  top_place_chain_key text,
  created_at timestamptz not null default now()
);

alter table public.healthy_nearby_sessions enable row level security;
create policy "service_role_full_access_healthy_nearby_sessions"
on public.healthy_nearby_sessions for all to service_role using (true) with check (true);

create index if not exists idx_hns_date on public.healthy_nearby_sessions(created_at);
create index if not exists idx_hns_area on public.healthy_nearby_sessions(area_key);

-- 3. Daily stats snapshot (cron populates this)
create table if not exists public.daily_stats (
  id bigint generated by default as identity primary key,
  stat_date date not null unique,
  total_users integer default 0,
  active_users integer default 0,
  scans_today integer default 0,
  healthy_nearby_searches integer default 0,
  meals_logged integer default 0,
  total_chain_items integer default 0,
  total_real_menu_items integer default 0,
  created_at timestamptz not null default now()
);

alter table public.daily_stats enable row level security;
create policy "service_role_full_access_daily_stats"
on public.daily_stats for all to service_role using (true) with check (true);
